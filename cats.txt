            // 1. Get the current value, removing all non-digit characters (including '.')

            // 2. Format the clean numerical value using the Indian comma system
            if (cleanValue) {
                // Use parseInt for whole numbers
                const numberValue = parseInt(cleanValue, 10);
                if (!isNaN(numberValue)) {
                    // For input, we want the number without the currency symbol
                    inputElement.value = indianFormatter.format(numberValue);
                } else {
                    inputElement.value = cleanValue; // Keep the raw input if it's incomplete
                }
            } else {
                inputElement.value = '';
            }
        }

        // --- 1. Firebase Initialization and Authentication ---
        // Helper for exponential backoff (for retrying auth)
        const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));

        async function initAuth(app, retryCount = 0) {
            const authInstance = getAuth(app);
            try {
                // Attempt to sign in with custom token or anonymously
                if (initialAuthToken) {
                    await signInWithCustomToken(authInstance, initialAuthToken);
                } else {
                    await signInAnonymously(authInstance);
                }
                return authInstance;
            } catch (error) {
                console.error(`Attempt ${retryCount + 1} failed to authenticate:`, error);
                if (retryCount < MAX_RETRIES) {
                    const delay = Math.pow(2, retryCount) * 1000 + (Math.random() * 1000); // 2s, 4s, 8s, etc. + jitter
                    userInfo.textContent = `Retrying connection in ${Math.round(delay / 1000)}s... (Attempt ${retryCount + 2}/${MAX_RETRIES + 1})`;
                    await sleep(delay);
                    return initAuth(app, retryCount + 1); // Recursive retry
                }
                throw new Error("Authentication failed after multiple retries.");
            }
        }

        // --- 4. Core Accounting Logic: Calculate Balances ---
        function calculateBalances(transactions) {
            const balances = {};
            const accountNames = Object.keys(ACCOUNTS);

            accountNames.forEach(name => {
                balances[name] = 0;
            });

            transactions.forEach(t => {
                // Amounts are now guaranteed to be integers
                const amount = parseInt(t.amount, 10);

                if (balances.hasOwnProperty(t.debitAccount)) {
                    balances[t.debitAccount] += amount;
                }
                if (balances.hasOwnProperty(t.creditAccount)) {
                    balances[t.creditAccount] -= amount;
                }
            });

            const finalBalances = {};
            for (const name in balances) {
                const netBalance = balances[name];
                const accountType = ACCOUNTS[name]?.type;
                const normalBalance = ACCOUNT_CONFIGS[accountType]?.normalBalance;
                
                let displayBalance = netBalance;

                if (normalBalance === 'Credit') {
                    displayBalance = -netBalance;
                }
                
                // Keep as whole numbers
                finalBalances[name] = Math.round(displayBalance);
            }
            return finalBalances;
        }

        // --- Import from CSV Function ---
        async function importDayBookFromCSV(event) {
            const file = event.target.files[0];
            if (!file) return;

            importStatusEl.classList.remove('hidden', 'text-red-400', 'text-green-400');
            importStatusEl.classList.add('text-yellow-400');
            importStatusEl.textContent = 'Reading file... Do not navigate away.';

            const reader = new FileReader();
            reader.onload = async (e) => {
                const csvText = e.target.result;
                const lines = csvText.trim().split('\n');
                
                if (lines.length < 2) {
                    setImportStatus('Error: CSV is empty or only contains headers.', 'red');
                    return;
                }
                
                // 1. Validate Headers
                const headers = lines[0].split(',').map(h => h.trim().replace(/"/g, ''));
                const expectedHeaders = ["Date", "Voucher Type", "Narration", "Debit Account", "Credit Account", "Amount (INR)"];
                
                if (headers.join(',') !== expectedHeaders.join(',')) {
                    setImportStatus('Error: Invalid CSV format. Headers must be: ' + expectedHeaders.join(', '), 'red');
                    return;
                }

                const newTransactions = [];
                const existingAccountNames = Object.keys(ACCOUNTS);

                // 2. Parse and Validate Data Lines
                for (let i = 1; i < lines.length; i++) {
                    const line = lines[i];
                    // Basic CSV parsing that handles quoted strings (like Narration)
                    const row = parseCSVRow(line);

                    if (row.length !== expectedHeaders.length) {
                        setImportStatus(`Error on line ${i + 1}: Expected ${expectedHeaders.length} columns, got ${row.length}.`, 'red');
                        return;
                    }

                    let [date, voucherType, narration, debitAccount, creditAccount, amountString] = row;
                    
                    // --- DATE FORMAT FIX: Handle DD-MM-YYYY and convert to YYYY-MM-DD ---
                    let dateForValidation = date;
                    if (date.match(/^\d{2}-\d{2}-\d{4}$/)) { // Checks for DD-MM-YYYY format
                        const parts = date.split('-');
                        // Convert to YYYY-MM-DD
                        dateForValidation = `${parts[2]}-${parts[1]}-${parts[0]}`;
                    }

                    const parsedDate = new Date(dateForValidation);
                    
                    // Validation Checks
                    if (isNaN(parsedDate.getTime()) || !dateForValidation.match(/^\d{4}-\d{2}-\d{2}$/)) {
                        setImportStatus(`Error on line ${i + 1}: Invalid Date format. Must be YYYY-MM-DD or DD-MM-YYYY.`, 'red');
                        return;
                    }

                    const amount = parseInt(amountString, 10);
                    
                    if (!VOUCHER_TYPES.includes(voucherType.replace(' Voucher', ''))) { // Remove " Voucher" for comparison
                        setImportStatus(`Error on line ${i + 1}: Invalid Voucher Type. Check spelling (e.g., Journal, Payment).`, 'red');
                        return;
                    }
                    if (isNaN(amount) || amount <= 0) {
                        setImportStatus(`Error on line ${i + 1}: Invalid or zero Amount.`, 'red');
                        return;
                    }
                    if (debitAccount === creditAccount) {
                         setImportStatus(`Error on line ${i + 1}: Debit and Credit accounts must be different.`, 'red');
                        return;
                    }
                    if (!existingAccountNames.includes(debitAccount) || !existingAccountNames.includes(creditAccount)) {
                        setImportStatus(`Error on line ${i + 1}: Account '${debitAccount}' or '${creditAccount}' not found in your Chart of Accounts. Please add it first.`, 'red');
                        return;
                    }

                    newTransactions.push({
                        // Store the validated YYYY-MM-DD format
                        date: dateForValidation,
                        voucherType: voucherType,
                        narration: narration,
                        debitAccount: debitAccount,
                        creditAccount: creditAccount,
                        amount: amount,
                        timestamp: new Date().toISOString()
                    });
                }
                
                // 3. Batch Write to Firestore
                if (newTransactions.length > 0) {
                    try {
                        const batch = writeBatch(db);
                        const collectionRef = getTransactionsCollectionRef(db, userId, appId);
                        
                        newTransactions.forEach(t => {
                            const docRef = doc(collectionRef); // Generate a new doc ref for each
                            batch.set(docRef, t);
                        });
                        
                        await batch.commit();
                        setImportStatus(`Success! Imported ${newTransactions.length} vouchers.`, 'green');
                        // The onSnapshot listener handles UI update
                        
                    } catch (e) {
                        console.error("Batch write failed:", e);
                        setImportStatus(`Error saving data to database: ${e.message}`, 'red');
                    }
                }
            };
            reader.onerror = () => {
                setImportStatus('Error reading file.', 'red');
            };
            
            reader.readAsText(file);
        }

        // Helper function for basic CSV row parsing (handles quoted narration)
        function parseCSVRow(text) {
            const result = [];
            let inQuotes = false;
            let currentField = '';

            for (let i = 0; i < text.length; i++) {
                const char = text[i];

                if (char === '"') {
                    if (inQuotes && text[i + 1] === '"') {
                        // Escaped double quote "" -> "
                        currentField += '"';
                        i++; // Skip next quote
                    } else {
                        // Toggle quote state
                        inQuotes = !inQuotes;
                    }
                } else if (char === ',' && !inQuotes) {
                    // End of field
                    result.push(currentField.trim().replace(/^"|"$/g, '')); // Trim and remove outer quotes if they exist
                    currentField = '';
                } else {
                    currentField += char;
                }
            }
            result.push(currentField.trim().replace(/^"|"$/g, '')); // Push the last field
            return result;
        }

        function renderLedger() {
            const selectedAccount = ledgerAccountSelect.value;
            ledgerContent.innerHTML = '';

            if (!selectedAccount) {
                 ledgerContent.innerHTML = '<p class="p-6 text-center text-gray-400">Please select an account above to view its detailed ledger.</p>';
                 return;
            }
            
            const accountType = ACCOUNTS[selectedAccount]?.type;
            const normalBalance = ACCOUNT_CONFIGS[accountType]?.normalBalance;
            const finalBalance = calculateBalances(ALL_TRANSACTIONS)[selectedAccount];

            // Filter and prepare ledger entries
            const ledgerEntries = ALL_TRANSACTIONS
                .filter(t => t.debitAccount === selectedAccount || t.creditAccount === selectedAccount)
                .map(t => {
                    const isDebit = t.debitAccount === selectedAccount;
                    const amount = t.amount;
                    // Ledger narration should show the CONTRA account
                    const contraAccount = isDebit ? t.creditAccount : t.debitAccount;
                    return {
                        ...t,
                        isDebit,
                        contraAccount
                    };
                })
                .sort((a, b) => new Date(a.date) - new Date(b.date) || new Date(a.timestamp) - new Date(a.timestamp));


            let runningBalance = 0;

            let html = `
                <h3 class="text-xl font-bold p-4 bg-gray-600">${selectedAccount} Ledger (Type: ${accountType})</h3>
                <table class="min-w-full divide-y divide-gray-600">
                    <thead class="bg-gray-700">
                        <tr>
                            <th class="px-4 py-3 text-left text-xs font-semibold text-gray-400 uppercase tracking-wider">Date</th>
                            <th class="px-4 py-3 text-left text-xs font-semibold text-gray-400 uppercase tracking-wider">Particulars</th>
                            <th class="px-4 py-3 text-right text-xs font-semibold text-gray-400 uppercase tracking-wider text-blue-400">Debit (₹)</th>
                            <th class="px-4 py-3 text-right text-xs font-semibold text-gray-400 uppercase tracking-wider text-red-400">Credit (₹)</th>
                            <th class="px-4 py-3 text-right text-xs font-semibold text-gray-400 uppercase tracking-wider">Balance (₹)</th>
                        </tr>
                    </thead>
                    <tbody class="bg-gray-800 divide-y divide-gray-700">
            `;
            
            // Render entries and calculate running balance
            ledgerEntries.forEach(entry => {
                const debit = entry.isDebit ? entry.amount : 0;
                const credit = entry.isDebit ? 0 : entry.amount;
                
                // Calculate running balance based on Normal Balance rule for the ledger
                if (entry.isDebit) {
                    runningBalance += (normalBalance === 'Debit') ? entry.amount : -entry.amount;
                } else { // is Credit
                    runningBalance += (normalBalance === 'Credit') ? entry.amount : -entry.amount;
                }

                const balanceText = formatter.format(Math.abs(runningBalance));
                const balanceColor = runningBalance >= 0 ? 'text-green-400' : 'text-red-400';
                const balanceType = runningBalance >= 0 ? normalBalance : (normalBalance === 'Debit' ? 'Credit' : 'Debit');

                html += `
                    <tr class="hover:bg-gray-700 transition duration-100">
                        <td class="px-4 py-3 whitespace-nowrap text-sm text-gray-200">${new Date(entry.date).toLocaleDateString('en-IN')}</td>
                        <td class="px-4 py-3 text-sm text-gray-300">
                            ${entry.isDebit ? 'To' : 'By'} ${entry.contraAccount} (${entry.voucherType})
                            <p class="text-xs text-gray-500">${entry.narration}</p>
                        </td>
                        <td class="px-4 py-3 whitespace-nowrap text-sm text-right text-blue-400">${debit > 0 ? formatter.format(debit) : ''}</td>
                        <td class="px-4 py-3 whitespace-nowrap text-sm text-right text-red-400">${credit > 0 ? formatter.format(credit) : ''}</td>
                        <td class="px-4 py-3 whitespace-nowrap text-sm text-right font-semibold ${balanceColor}">
                            ${balanceText} <span class="text-xs text-gray-500">(${balanceType})</span>
                        </td>
                    </tr>
                `;
            });

            html += `
                    </tbody>
                </table>
                <div class="p-4 bg-gray-600 font-extrabold text-lg flex justify-between rounded-b-lg">
                    <span>FINAL CLOSING BALANCE:</span>
                    <span class="${finalBalance >= 0 ? 'text-green-400' : 'text-red-400'}">${formatter.format(Math.abs(finalBalance))} (${finalBalance >= 0 ? normalBalance : (normalBalance === 'Debit' ? 'Credit' : 'Debit')})</span>
                </div>
            `;
            
            ledgerContent.innerHTML = html;
        }

        function renderTrialBalance(balances) {
            const reportEl = document.getElementById('trial-balance-report');

                            <td class="pt-2 pb-1 text-gray-300">Account</td>

                        </tr>
                    </thead>
                    <tbody>
            `;

            const accountNames = Object.keys(ACCOUNTS).sort();

            accountNames.forEach(name => {
                const balance = balances[name];
                const accountType = ACCOUNTS[name]?.type;
                const normalBalance = ACCOUNT_CONFIGS[accountType]?.normalBalance;
                const displayValue = Math.abs(balance);

                let debit = 0;
                let credit = 0;

                // Trial Balance Logic
                if (balance > 0) {
                    if (normalBalance === 'Debit') {
                        debit = displayValue;
                        totalDebit += debit;
                    } else { // Normal Balance is Credit
                        credit = displayValue;
                        totalCredit += credit;
                    }
                }
                else if (balance < 0) {
                    if (normalBalance === 'Debit') {
                        credit = displayValue;
                        totalCredit += credit;
                    } else { // Normal Balance is Credit
                        debit = displayValue;
                        totalCredit += debit;
                    }
                }


                if (debit > 0 || credit > 0) {
                    html += `
                        <tr class="border-b border-gray-700 hover:bg-gray-700">
                            <td class="py-1 text-gray-300">${name}</td>
                            <td class="text-right text-gray-300">${debit > 0 ? formatter.format(debit) : ''}</td>
                            <td class="text-right text-gray-300">${credit > 0 ? formatter.format(credit) : ''}</td>

                }

            html += `
                    </tbody>
                    <tfoot class="font-extrabold bg-gray-600 text-gray-100">
                        <tr>
                            <td class="py-2">TOTALS</td>
                            <td class="text-right py-2">${formatter.format(totalDebit)}</td>
                            <td class="text-right py-2">${formatter.format(totalCredit)}</td>

        // --- 6. Voucher Submission ---
        form.addEventListener('submit', addTransaction);

        async function addTransaction(event) {
            event.preventDefault();

            if (!isAuthReady || !userId) {
                errorMsg.textContent = 'Authentication not ready. Please wait.';
                errorMsg.classList.remove('hidden');
                return;
            }

            submitBtn.disabled = true;
            errorMsg.classList.add('hidden');
            transactionSuccessMsg.classList.add('hidden'); // Hide on submission attempt

            const date = document.getElementById('date').value;
            const voucherType = voucherTypeSelect.value;
            // Changed from description to narration
            const narration = narrationInput.value.trim(); 
            const debitAccount = debitAccountSelect.value;
            const creditAccount = creditAccountSelect.value;
            
            // Get the formatted string amount and clean it up for calculation/storage
            const amountInput = document.getElementById('amount').value;
            // Strip ALL non-digit characters for whole number input
            const amountCleaned = amountInput.replace(/[^0-9]/g, '');
            const amount = parseInt(amountCleaned, 10); // Use parseInt for whole numbers


            if (debitAccount === creditAccount) {
                errorMsg.textContent = 'Debit and Credit accounts must be different.';
                errorMsg.classList.remove('hidden');
                submitBtn.disabled = false;
                return;
            }
            if (amount <= 0 || isNaN(amount)) {
                errorMsg.textContent = 'Amount must be a positive whole number.';
                errorMsg.classList.remove('hidden');
                submitBtn.disabled = false;
                return;
            }

            const transactionData = {
                date: date,
                voucherType: voucherType,
                narration: narration, // Using narration key
                debitAccount: debitAccount,
                creditAccount: creditAccount,
                amount: amount, // Store the cleaned integer amount
                timestamp: new Date().toISOString()
            };

            try {
                const collectionRef = getTransactionsCollectionRef(db, userId, appId);
                await addDoc(collectionRef, transactionData);
                
                // --- POST-SUBMISSION LOGIC ---
                // 1. Clear ONLY the sensitive, specific fields (amount and accounts)
                debitAccountSelect.value = '';
                creditAccountSelect.value = '';
                document.getElementById('amount').value = ''; 
                
                // 2. KEEP date, voucherType, and narration
                
                // 3. Save the current form state (with empty accounts/amount) to session
                saveFormToSession(); 

                // Show success notification
                transactionSuccessMsg.classList.remove('hidden');
                setTimeout(() => {
                    transactionSuccessMsg.classList.add('hidden');
                }, 3000); 

                // Switch to Day Book view after successful entry
                switchView('DayBook');

            } catch (e) {
                console.error("Error recording transaction: ", e);
                errorMsg.textContent = `Error recording transaction: ${e.message}`;
                errorMsg.classList.remove('hidden');
            } finally {
                submitBtn.disabled = false;
            }
        }

        // --- 7. Real-time Listener for Transactions ---
        function listenForTransactions() {
            if (!isAuthReady || !userId) return;

            const collectionRef = getTransactionsCollectionRef(db, userId, appId);

            onSnapshot(collectionRef, (snapshot) => {
                const transactions = [];
                snapshot.forEach((doc) => {
                    // Check if narration exists or description exists (for older entries)
                    let data = doc.data();
                    if (!data.narration && data.description) {
                         data.narration = data.description;
                         delete data.description;
                    }
                    transactions.push({ id: doc.id, ...data });
                });
                updateUI(transactions);
            }, (error) => {
                console.error("Error listening to transactions: ", error);
            });
        }
